#include <stdio.h>

#define N 3  /* numero de procesos */

int main(void) {
    /* Ejemplo del libro de Silberschatz: P1=24, P2=3, P3=3 */
    int rafaga[N] = {24, 3, 3};   /* tiempo de CPU que necesita cada proceso */
    int restante[N];              /* tiempo que le falta a cada proceso */
    int espera[N];                /* tiempo de espera */
    int retorno[N];               /* tiempo de retorno (turnaround) */
    int quantum = 4;
    int tiempo = 0;               /* reloj del sistema */
    int terminados = 0;

    for (int i = 0; i < N; i++) {
        restante[i] = rafaga[i];
        espera[i] = 0;
        retorno[i] = 0;
    }

    printf("Quantum = %d ms\n\n", quantum);
    printf("Diagrama de ejecucion:\n");

    /* Se recorre la cola de forma circular hasta que todos terminen.
       Todos los procesos llegan en t = 0. */
    while (terminados < N) {
        for (int i = 0; i < N; i++) {
            if (restante[i] > 0) {
                /* Ejecuta un quantum completo o lo que le quede */
                int ejecucion = (restante[i] < quantum) ? restante[i] : quantum;

                printf("  t=%2d -> t=%2d : P%d\n", tiempo, tiempo + ejecucion, i + 1);

                tiempo += ejecucion;
                restante[i] -= ejecucion;

                if (restante[i] == 0) {          /* el proceso termino */
                    retorno[i] = tiempo;
                    espera[i] = retorno[i] - rafaga[i];
                    terminados++;
                }
            }
        }
    }

    /* Resultados */
    float total_espera = 0, total_retorno = 0;
    printf("\n%-9s %-8s %-8s %-8s\n", "Proceso", "Rafaga", "Espera", "Retorno");
    for (int i = 0; i < N; i++) {
        printf("P%-8d %-8d %-8d %-8d\n", i + 1, rafaga[i], espera[i], retorno[i]);
        total_espera += espera[i];
        total_retorno += retorno[i];
    }

    printf("\nEspera promedio:  %.2f ms\n", total_espera / N);
    printf("Retorno promedio: %.2f ms\n", total_retorno / N);

    return 0;
}
